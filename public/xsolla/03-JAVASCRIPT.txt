/* ISOCORE BROADCAST — Xsolla JavaScript
   Clean live edition: no temporary on-screen debug panel.
   Xsolla CSS controls the 500 x 350 camera and website placement.
*/
(function () {
  'use strict';

  var HOST = 'https://isocore-broadcast.tgvain1.workers.dev';
  var VERSION = '16';

  function load(file) {
    return new Promise(function (resolve, reject) {
      var script = document.createElement('script');
      script.src = HOST + '/assets/' + file + '?v=' + VERSION;
      // Load a normal classic script; do not force CORS mode.
      script.onload = resolve;
      script.onerror = function () {
        reject(new Error('Isocore could not load ' + file));
      };
      document.head.appendChild(script);
    });
  }

  function boot() {
    var root = document.getElementById('isocore-broadcast');
    if (!root) {
      console.error('[ISOCORE] HTML container not found');
      return;
    }
    if (root.dataset.xsollaBooted === '1') return;
    root.dataset.xsollaBooted = '1';
    document.body.classList.add('isocore-lightstream');

    load('stage.js')
      .then(function () { return load('overlay-stage.js'); })
      .then(function () {
        return load('lightstream-dialogue.js').catch(function () {
          console.warn('[ISOCORE] Optional subtitles unavailable');
        });
      })
      .then(function () { return load('bridge.js'); })
      .catch(function (error) {
        console.error('[ISOCORE]', error);
        root.dataset.loadError = error.message;
      });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', boot, { once: true });
  } else {
    boot();
  }
})();
