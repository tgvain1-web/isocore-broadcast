/* ISOCORE BROADCAST — Xsolla Code / JavaScript tab.
 * Paste ONLY this JS, without <script> tags.
 * The 3D engine & art are fetched from the independent Cloudflare Worker.
 * Never include control tokens, OAuth credentials, or stream keys.
 */
(function(){
  'use strict';
  var HOST='https://REPLACE_WITH_YOUR_DEPLOYED_ISOCORE_BROADCAST_HOST';
  var VERSION='14';
  function boot(){
    var root=document.getElementById('isocore-broadcast');
    if(!root || root.dataset.xsollaBooted==='1')return;
    root.dataset.xsollaBooted='1';
    document.body.classList.add('isocore-lightstream');
    var q=new URLSearchParams(location.search);
    if(q.get('debug')==='1')document.body.classList.add('isocore-lightstream-debug');
    if(q.get('layout')==='large')document.body.classList.add('isocore-lightstream-large');
    var diag=document.getElementById('iso-diag');
    function error(msg){
      if(diag){diag.textContent='ISOCORE // '+msg;diag.classList.add('error');}
      // Do not show a startup error over PS5 gameplay; keep errors in developer console.
      try{console.error('[Isocore Broadcast] '+msg);}catch(e){}
    }
    try{
      var u=new URL(HOST);
      if(u.protocol!=='https:'||!u.host||u.hostname.indexOf('replace_with_')>=0)throw Error('Set your real Cloudflare Worker HTTPS URL');
    }catch(e){error(e.message);return;}
    function load(name){
      return new Promise(function(resolve,reject){
        var script=document.createElement('script');
        script.src=HOST+'/assets/'+name+'?v='+VERSION;
        script.crossOrigin='anonymous';
        script.onload=resolve;
        script.onerror=function(){reject(Error('Script unavailable or Xsolla CSP blocks '+name))};
        document.head.appendChild(script);
      });
    }
    load('stage.js')
      .then(function(){return load('overlay-stage.js')})
      .then(function(){return load('lightstream-dialogue.js').catch(function(){return null;})})
      .then(function(){return load('bridge.js')})
      .catch(function(e){error(e.message)});
  }
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',boot,{once:true});
  else boot();
})();
