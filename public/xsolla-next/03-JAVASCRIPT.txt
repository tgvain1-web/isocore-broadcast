/* ISOCORE 3D REVIEW — copy into Xsolla JavaScript tab.
   Uses isolated GitHub review models; does not change live broadcasting. */
(function(){
  "use strict";
  const BASE="https://cdn.jsdelivr.net/gh/tgvain1-web/isocore-broadcast@2cb66cc88aeca1f2f7f810ae166a7b719f901f02/public/";
  const root=document.getElementById("isocore-broadcast");
  const status=document.getElementById("iso-review-status");
  if(!root||root.dataset.reviewBooted)return;
  root.dataset.reviewBooted="1";
  let desired=["ghost","stage-front"], ready=false, stopped=false;
  document.body.classList.add("isocore-lightstream","isocore-next-world");
  function say(msg){if(status)status.textContent=msg;}
  function load(path){
    return new Promise((resolve,reject)=>{
      const s=document.createElement("script");
      s.src=BASE+path;
      s.onload=resolve;
      s.onerror=()=>reject(Error("Failed to load "+path));
      document.head.appendChild(s);
    });
  }
  function style(path){
    const link=document.createElement("link");link.rel="stylesheet";
    link.href=BASE+path;document.head.appendChild(link);
  }
  function choose(who,shot){
    desired=[who,shot];
    document.querySelectorAll("#iso-review-views button").forEach(btn=>{
      btn.setAttribute("aria-pressed",String(btn.dataset.who===who&&btn.dataset.shot===shot));
    });
    updateCamera();
  }
  // 16:9 Isocore stage at the origin, equipment at x=-16 and +16.
  // The review camera is intentionally independent of the live broadcast framing.
  const CAMERA_VIEWS = {
    "stage-front": [0, 9.5, 24.5, 0, 2.8, 0],
    "stage-back":  [0, 9.5, -24.5, 0, 2.8, 0],
    "high":        [0, 18, 20, 0, 11.5, 0]
  };
  let lastCameraKey="";
  function updateCamera(){
    const stage=document.getElementById("ig-ghost-world");
    const engine=stage&&stage._igw;
    if(!engine||typeof engine.broadcast!=="function")return;
    const who=desired[0],shot=desired[1],cam=CAMERA_VIEWS[shot];
    const key=who+":"+shot;
    if(ready&&lastCameraKey===key)return;
    try{
      // Enter the existing world suite to ensure both rigs and avatars are visible.
      if(!engine.broadcast(who,cam?"wide":shot))return;
      if(typeof engine.filmCam==="function")engine.filmCam(cam||null);
      ready=true;lastCameraKey=key;
      say(cam ? ("Showing "+shot.replace("-"," ").toUpperCase()+" · 1920×1080 REVIEW") :
        ("Showing "+who.toUpperCase()+" · 1920×1080 REVIEW"));
    }catch(e){say("Camera error: "+e.message);}
  }
  document.querySelectorAll("#iso-review-views button").forEach(btn=>{
    btn.addEventListener("click",()=>choose(btn.dataset.who,btn.dataset.shot));
  });
  window.addEventListener("error",e=>{
    if(/module|cors|webgl|fetch|import/i.test(e.message||""))say("Preview error: "+e.message);
  });
  async function boot(){
    try{
      say("Loading Isocore 3D engine…");
      style("assets/ig-ghost-world.css");
      await load("assets/stage.js");
      if(!window.ISOCORE_BROADCAST_STAGE)throw Error("Stage HTML unavailable");
      root.innerHTML='<div id="iso-camera"><div id="igwHomeSlot"></div><div id="iso-camera-label"></div></div><div id="iso-overlay-shell"></div><div id="iso-diag"></div>';
      document.getElementById("igwHomeSlot").innerHTML=window.ISOCORE_BROADCAST_STAGE;
      await load("assets/ig-ghost-world-next.js");
      choose("ghost","stage-front");
      let attempts=0;
      const poll=setInterval(()=>{
        if(stopped){clearInterval(poll);return;}
        updateCamera();
        if(++attempts>180){
          clearInterval(poll);
          if(!ready)say("Preview not ready: check WebGL or network access");
        }
      },1000);
    }catch(e){say("Preview failed: "+e.message);console.error(e);}
  }
  if(document.readyState==="loading")document.addEventListener("DOMContentLoaded",boot,{once:true});
  else boot();
})();
